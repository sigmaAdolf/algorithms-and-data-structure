## 2. Теоретическая часть
**Быстрая сортировка (Quick Sort)** — алгоритм типа «разделяй и властвуй». Выбирает опорный элемент, разбивает массив на две части (меньше и больше опорного) и рекурсивно сортирует их. В среднем работает за O(n log n), в худшем — за O(n²).

**Сортировка слиянием (Merge Sort)** — также использует принцип «разделяй и властвуй», но всегда делит массив строго пополам и затем сливает отсортированные половины. Всегда работает за O(n log n), но требует O(n) дополнительной памяти.

**Пирамидальная сортировка (Heap Sort)** — использует структуру данных «двоичная куча» (binary heap). Работает за O(n log n) и не требует дополнительной памяти.

**OpenMP** — технология для распараллеливания кода на многоядерных процессорах. Позволяет выполнять независимые участки кода (например, сортировку левой и правой половин массива) одновременно.

## 4. Листинг программы
```cpp
#include <iostream>
#include <chrono>
#include <algorithm>
#include <cstdlib>
#include <omp.h>

using namespace std;
using namespace chrono;

int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = low - 1;
    for (int j = low; j < high; j++) {
        if (arr[j] <= pivot) {
            i++;
            swap(arr[i], arr[j]);
        }
    }
    swap(arr[i + 1], arr[high]);
    return i + 1;
}

void quickSortSequential(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quickSortSequential(arr, low, pi - 1);
        quickSortSequential(arr, pi + 1, high);
    }
}

void quickSortParallel(int arr[], int low, int high, int depth = 0) {
    if (low < high) {
        int pi = partition(arr, low, high);

        if (depth < 4) {
#pragma omp parallel sections
            {
#pragma omp section
                { quickSortParallel(arr, low, pi - 1, depth + 1); }

#pragma omp section
                { quickSortParallel(arr, pi + 1, high, depth + 1); }
            }
        }
        else {
            quickSortParallel(arr, low, pi - 1, depth + 1);
            quickSortParallel(arr, pi + 1, high, depth + 1);
        }
    }
}

void merge(int arr[], int left, int mid, int right) {
    int n1 = mid - left + 1;
    int n2 = right - mid;
    int* L = new int[n1];
    int* R = new int[n2];

    for (int i = 0; i < n1; i++) L[i] = arr[left + i];
    for (int j = 0; j < n2; j++) R[j] = arr[mid + 1 + j];

    int i = 0, j = 0, k = left;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) arr[k++] = L[i++];
        else arr[k++] = R[j++];
    }
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];

    delete[] L;
    delete[] R;
}

void mergeSort(int arr[], int left, int right) {
    if (left < right) {
        int mid = left + (right - left) / 2;
        mergeSort(arr, left, mid);
        mergeSort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}

void heapify(int arr[], int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n && arr[left] > arr[largest]) largest = left;
    if (right < n && arr[right] > arr[largest]) largest = right;

    if (largest != i) {
        swap(arr[i], arr[largest]);
        heapify(arr, n, largest);
    }
}

void heapSort(int arr[], int n) {
    for (int i = n / 2 - 1; i >= 0; i--) heapify(arr, n, i);
    for (int i = n - 1; i > 0; i--) {
        swap(arr[0], arr[i]);
        heapify(arr, i, 0);
    }
}

int main() {
    setlocale(LC_ALL, "ru");

    const int N = 500000;
    int* original = new int[N];
    for (int i = 0; i < N; i++) original[i] = rand() % N;

    cout << "=== Сравнение сортировок (N = " << N << ") ===" << endl;

    int* arr1 = new int[N];
    copy(original, original + N, arr1);
    auto start1 = high_resolution_clock::now();
    quickSortSequential(arr1, 0, N - 1);
    auto end1 = high_resolution_clock::now();
    cout << "Быстрая (обычная):       " << duration_cast<milliseconds>(end1 - start1).count() << " мс" << endl;
    delete[] arr1;

    int* arr2 = new int[N];
    copy(original, original + N, arr2);
    auto start2 = high_resolution_clock::now();
    quickSortParallel(arr2, 0, N - 1);
    auto end2 = high_resolution_clock::now();
    cout << "Быстрая (параллельная):  " << duration_cast<milliseconds>(end2 - start2).count() << " мс" << endl;
    delete[] arr2;

    int* arr3 = new int[N];
    copy(original, original + N, arr3);
    auto start3 = high_resolution_clock::now();
    mergeSort(arr3, 0, N - 1);
    auto end3 = high_resolution_clock::now();
    cout << "Сортировка слиянием:     " << duration_cast<milliseconds>(end3 - start3).count() << " мс" << endl;
    delete[] arr3;

    int* arr4 = new int[N];
    copy(original, original + N, arr4);
    auto start4 = high_resolution_clock::now();
    heapSort(arr4, N);
    auto end4 = high_resolution_clock::now();
    cout << "Пирамидальная:           " << duration_cast<milliseconds>(end4 - start4).count() << " мс" << endl;
    delete[] arr4;

    delete[] original;
    return 0;
}
```

## 8. Ответы на контрольные вопросы

### 1. Какова сложность быстрой сортировки в лучшем, среднем и худшем случае?
Лучший и средний: **O(n log n)**. Худший: **O(n²)**.

### 2. Когда возникает худший случай для быстрой сортировки?
Худший случай возникает, когда опорный элемент выбирается крайне неудачно (например, всегда минимальный или максимальный в массиве). Это происходит, если массив уже отсортирован, а опорным выбирается первый или последний элемент. В этом случае массив разбивается на части размера 1 и n-1, что дает O(n²).

### 3. Как можно улучшить быструю сортировку для защиты от худшего случая?
Использовать выбор опорного элемента как **медианы трех** (первый, средний и последний элементы) или выбирать случайный элемент. Также помогает 3-частное разделение (для массивов с большим количеством дубликатов).

### 4. Какова сложность сортировки слиянием? Почему она всегда O(n log n)?
**O(n log n)**. Алгоритм всегда делит массив пополам (это log n уровней), и на каждом уровне происходит процесс слияния, который занимает O(n) времени. В отличие от быстрой сортировки, разбиение всегда равномерное, поэтому худшего случая O(n²) не бывает.

### 5. Какие преимущества и недостатки имеет сортировка слиянием перед быстрой?
**Преимущества:** Стабильность, гарантированное время O(n log n) в любом случае. Отлично подходит для сортировки связных списков и внешней сортировки (файлов).
**Недостатки:** Требует O(n) дополнительной памяти, на практике часто работает медленнее быстрой сортировки из-за выделения памяти и копирования.

### 6. Как работает пирамидальная сортировка? Какова её сложность?
Сначала массив превращается в двоичную кучу (max-heap), где родитель больше своих потомков. Затем самый большой элемент (корень) меняется местами с последним, и куча восстанавливается (heapify) для оставшейся части. Сложность: **O(n log n)** во всех случаях.

### 7. Какая из трёх сортировок устойчивая? Почему?
**Сортировка слиянием**. При слиянии двух половин, если элементы равны, мы сначала берем элемент из левой половины (`L[i] <= R[j]`), сохраняя их исходный относительный порядок. Быстрая и пирамидальная сортировки меняют элементы местами через большие расстояния, нарушая порядок.

### 8. Какая из трёх сортировок требует дополнительной памяти? Сколько?
**Сортировка слиянием** требует **O(n)** дополнительной памяти для временных массивов `L` и `R` при слиянии. Быстрая требует O(log n) памяти на стек рекурсии. Пирамидальная — O(1) (сортирует на месте).

### 9. Какая из трёх сортировок лучше работает на почти отсортированных данных?
**Быстрая сортировка** (если опорный элемент выбирается правильно, например, медиана) или **сортировка вставками** (если она используется как гибрид). Обычная быстрая сортировка с последним опорным элементом на почти отсортированных данных упадет в O(n²). Сортировка слиянием будет работать стабильно O(n log n).

### 10. Зачем в пирамидальной сортировке сначала строится куча?
Чтобы обеспечить свойство max-heap (родитель больше детей). Это гарантирует, что корень кучи всегда содержит **максимальный элемент** массива. Затем мы просто забираем этот максимум и ставим в конец, восстанавливая кучу для оставшихся элементов.

### 11. Что такое «разделяй и властвуй»? Как он применяется в быстрой сортировке и сортировке слиянием?
Это стратегия, при которой задача разбивается на подзадачи меньшего размера, которые решаются независимо, а затем их результаты объединяются. 
В Quick Sort: разделение по опорному элементу, рекурсивная сортировка частей (слияние не нужно). 
В Merge Sort: деление пополам, рекурсивная сортировка, слияние отсортированных частей.

### 12. Можно ли реализовать быструю сортировку итеративно (без рекурсии)?
Да, используя собственный стек (массив или `std::stack`) для хранения границ подмассивов, которые еще нужно отсортировать. Это позволяет избежать переполнения стека вызовов при большой глубине рекурсии.

### 13. Почему быстрая сортировка обычно быстрее сортировки слиянием на практике?
Из-за лучшей локальности кэша (работает с данными в пределах одного массива, не выделяет новую память) и меньшего количества операций копирования. Сортировка слиянием тратит время на выделение памяти и копирование элементов туда-обратно.

### 14. Когда выгодно использовать пирамидальную сортировку?
Когда критически важна **память** (она работает in-place, O(1)), и когда нельзя допустить худшего случая O(n²). Например, в системах реального времени или на встроенных устройствах.

### 15. Какой алгоритм сортировки вы бы выбрали для встроенных систем с ограниченной памятью?
**Пирамидальную сортировку**. Она гарантирует время O(n log n) в любом случае и не требует дополнительной памяти (в отличие от слияния). Быстрая сортировка тоже не требует много памяти, но может упасть в O(n²), что недопустимо для критических систем.
